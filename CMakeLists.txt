cmake_minimum_required(VERSION 3.21)
project(mdownXTwitter VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
qt_add_executable(mdownXTwitter WIN32 src/main.cpp src/mainwindow.cpp src/mainwindow.h src/downloadengine.cpp src/downloadengine.h)
target_link_libraries(mdownXTwitter PRIVATE Qt6::Widgets)
target_compile_definitions(mdownXTwitter PRIVATE QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)
if(MSVC)
  target_compile_options(mdownXTwitter PRIVATE /utf-8 /W4)
else()
  target_compile_options(mdownXTwitter PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS mdownXTwitter RUNTIME DESTINATION .)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  qt_add_executable(MockDownloadEngine tests/mockengine.cpp)
  target_link_libraries(MockDownloadEngine PRIVATE Qt6::Core)
  qt_add_executable(EngineTests tests/enginetests.cpp src/downloadengine.cpp src/downloadengine.h)
  target_include_directories(EngineTests PRIVATE src)
  target_link_libraries(EngineTests PRIVATE Qt6::Test)
  add_dependencies(EngineTests MockDownloadEngine)
  add_test(NAME engine COMMAND EngineTests -o engine-tests.txt,txt -o -,txt)
  set_tests_properties(engine PROPERTIES ENVIRONMENT "MDOWN_TEST_ENGINE=$<TARGET_FILE:MockDownloadEngine>" TIMEOUT 30)
  qt_add_executable(UiTests tests/uitests.cpp src/mainwindow.cpp src/mainwindow.h src/downloadengine.cpp src/downloadengine.h)
  target_include_directories(UiTests PRIVATE src)
  target_link_libraries(UiTests PRIVATE Qt6::Test Qt6::Widgets)
  add_dependencies(UiTests MockDownloadEngine)
  add_test(NAME ui COMMAND UiTests -o ui-tests.txt,txt -o -,txt)
  set_tests_properties(ui PROPERTIES ENVIRONMENT "MDOWN_TEST_ENGINE=$<TARGET_FILE:MockDownloadEngine>;QT_QPA_PLATFORM=offscreen;QT_QPA_FONTDIR=C:/Windows/Fonts" TIMEOUT 30)
endif()
